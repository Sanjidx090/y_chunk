CHUNK MODE COMPARISON
════════════════════════════════════════════════════════════════════════

Example: 10-minute video (600 seconds)

MODE: "full"
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
├────────────────────────────────────────────────────────┤
0s                    600s (entire transcript)           
Output: 1 file


MODE: "fixed" with 30s chunks
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
├──────┼──────┼──────┼──────┼──────┼──────┼──────┼───...
0-30s  30-60s 60-90s 90-120s ...                  570-600s
Output: 20 files (chunks)


MODE: "sentences"
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
├┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┼┤
Each subtitle segment = 1 chunk (typically 2-5s each)
Output: ~200-300 files


MODE: "custom" with VAD [(0,45), (60,180), (200,450)]
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Output: 3 files (only spoken parts)
